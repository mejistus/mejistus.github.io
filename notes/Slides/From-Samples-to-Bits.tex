\begin{frame}{Sampling in the frequency domain}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item $x$ is \textbf{band-limited}: $X(f)=0$ for $|f|>B$.
        \item Sampling multiplies by a Dirac comb:
          \[ x_s(t)=x(t)\sum_n \delta(t-nT) \]
        \item So the spectrum repeats every $f_s$:
          \[ X_s(f)=f_s\sum_k X(f-kf_s) \]
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{center}

        \node[anchor=west] at (0.9,0.9) {$X(f)$};
        \begin{scope}[yshift=-2.2cm]

\begin{frame}{The sampling theorem}
  \begin{block}{Theorem (Nyquist 1928, Shannon 1949 \cite{nyquist1928,shannon1949})}
    If $X(f)=0$ for $|f|>B$ and $f_s>2B$, the samples $x(nT)$ determine $x(t)$ exactly.
  \end{block}
  \begin{center}
  \begin{tikzpicture}[font=\small, >={Stealth[length=1.6mm]}]
    \draw[->] (-2.6,0) -- (2.6,0) node[right] {$f$};
    \foreach \k in {-1,1}
      \filldraw[soft, fill=soft!18, thick] ({1.9*\k-0.7},0) -- ({1.9*\k},1) -- ({1.9*\k+0.7},0);
    \filldraw[sig, fill=sig!20, thick] (-0.7,0) -- (0,1) -- (0.7,0);
    \node[below] at (1.9,0) {$f_s$};
    \node[below] at (0,0) {$0$};
    \node[align=center] at (0,-0.85) {$f_s>2B$: copies stay apart};
    \begin{scope}[xshift=6.8cm]
      \draw[->] (-2.6,0) -- (2.6,0) node[right] {$f$};
      \foreach \k in {-1,1}
        \filldraw[soft, fill=soft!18, thick] ({1.0*\k-0.7},0) -- ({1.0*\k},1) -- ({1.0*\k+0.7},0);
      \filldraw[sig, fill=sig!20, thick] (-0.7,0) -- (0,1) -- (0.7,0);
      \fill[hot, opacity=0.75] (0.3,0) -- (0.5,0.2857) -- (0.7,0) -- cycle;
      \fill[hot, opacity=0.75] (-0.3,0) -- (-0.5,0.2857) -- (-0.7,0) -- cycle;
      \node[below] at (1.0,0) {$f_s$};
      \node[below] at (0,0) {$0$};
      \node[align=center] at (0,-0.85) {$f_s<2B$: copies overlap, \textcolor{hot}{aliasing}};
    \end{scope}
  \end{tikzpicture}
  \end{center}
  $2B$ is the \textbf{Nyquist rate}. Whittaker (1915) and Kotelnikov (1933) found it too.
\end{frame}

\begin{frame}{Reconstruction: sinc interpolation}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \begin{itemize}
        \item Keep the central copy with an ideal low-pass filter at $f_s/2$.
        \item In time, that filter is a sinc: $\operatorname{sinc}(u)=\frac{\sin\pi u}{\pi u}$.
        \item Each sample adds one sinc, which is $0$ at all other sample points.
        \item In practice: low-pass \emph{before} sampling (CD: 44.1\,kHz for $\approx$20\,kHz), and approximate the sinc (linear, cubic, Lanczos).
      \end{itemize}
    \end{column}
    \begin{column}{0.56\textwidth}

\begin{frame}{Images are 2-D samples}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{itemize}
        \item A pixel grid samples the scene; frequencies are in \emph{cycles per pixel} $(u,v)$.
        \item Nyquist limit: $|u|,|v|<\tfrac12$, at least 2 pixels per period. Finer detail aliases: moir\'e, jaggies.

\begin{frame}{Why compress, and how far can we go?}
  A 12-megapixel photo is $4000\times3000\times3$ bytes $=36$\,MB raw; as a JPEG, a few MB.
  \begin{center}
  \begin{tikzpicture}[font=\small, >={Stealth[length=2mm]}, node distance=9mm,
      box/.style={draw, rounded corners=2pt, minimum height=9mm, minimum width=22mm, align=center}]
    \node (x) {image $x$};
    \node[box, right=of x] (t) {transform};
    \node[box, right=of t, draw=hot, very thick] (q) {quantize};
    \node[box, right=of q] (e) {entropy code};
    \node[right=of e, font=\ttfamily] (b) {0110...};
    \draw[->] (x) -- (t);
    \draw[->] (t) -- (q);
    \draw[->] (q) -- (e);
    \draw[->] (e) -- (b);
    \node[below=2mm of t, soft] {invertible};
    \node[below=2mm of q, hot] {the only lossy step};
    \node[below=2mm of e, soft] {invertible};
  \end{tikzpicture}
  \end{center}
  \begin{itemize}
    \item \textbf{Redundancy}: neighbouring pixels are correlated. Transform, then entropy-code.
      Lossless limit: the entropy $H$ \cite{shannon1948}.
    \item \textbf{Irrelevance}: detail the eye does not see. Quantize it coarsely.
      Lossy limit: the rate--distortion function $R(D)$ \cite{shannon1959}.
  \end{itemize}
\end{frame}

% 0:50
\begin{frame}{The limits: entropy and rate--distortion}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \textbf{Lossless}: at least the entropy $H=-\sum_i p_i\log_2 p_i$ bits per symbol \cite{shannon1948}.
        \item \textbf{Lossy}: $R(D)$, the fewest bits per sample at mean distortion $\le D$ \cite{shannon1959}.
        \item Gaussian source, MSE:
          \[ R(D)=\tfrac12\log_2\frac{\sigma^2}{D}, \quad D=\sigma^2\,2^{-2R} \]
          Each extra bit per sample: error $\div 4$, \textbf{$+6$\,dB}.
        \item A real codec is a point above the curve; 70 years of codecs = moving towards it.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{center}
      \begin{tikzpicture}
        \begin{axis}[width=7cm, height=5.2cm, axis lines=left, xmin=0, xmax=3, ymin=0, ymax=1.1,
            xlabel={rate $R$ (bits/sample)}, ylabel={$D/\sigma^2$}, ytick={0,0.5,1}, xtick={0,1,2,3},
            domain=0:3, samples=80, no markers,

      \end{center}

  \end{columns}
\end{frame}

\begin{frame}{Why the DCT? Energy compaction}

\begin{frame}{Link 1: count the numbers, then the bits}
  \begin{block}{Shannon (1949) \cite{shannon1949}}
    A signal of bandwidth $W$ lasting $T$ seconds is fixed by about $2WT$ numbers:
    a point in a $2WT$-dimensional space.
  \end{block}
  \begin{itemize}
    \item The same paper that proves the sampling theorem uses it to count dimensions, and then computes the channel capacity $W\log_2(1+S/N)$.
    \item Sampling turns a waveform into numbers \emph{without loss}; rate--distortion theory then prices each number in bits.
    \item For an image: pixels $\times$ bits per pixel. A $4000\times3000$ photo at 1 bit/pixel is 1.5\,MB.
    \item But the best split between the two factors depends on the budget: \alert{next slide}.
  \end{itemize}
\end{frame}

\begin{frame}{Link 2: the optimal lossy code is a low-pass filter}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \textbf{Reverse water-filling} for a Gaussian process with spectrum $S(f)$ \cite{berger1971}:
      \[ D_\theta=\int\!\min\{\theta,S(f)\}\,df \]
      \[ R_\theta=\int\!\tfrac12\log_2^{+}\frac{S(f)}{\theta}\,df \]
      \begin{itemize}
        \item Frequencies with $S(f)<\theta$ get \alert{zero bits}: they are dropped.
        \item Natural images: $S\propto 1/f^2$ \cite{field1987}. Fewer bits $\Rightarrow$ higher $\theta$ $\Rightarrow$ narrower band.
        \item JPEG's tables and zeroed DCT coefficients do this by hand.
      \end{itemize}
    \end{column}
    \begin{column}{0.54\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=1.55cm, y=1.9cm, font=\small, >={Stealth[length=1.6mm]},

\begin{frame}{Open questions}
  \begin{itemize}
    \item \textbf{Joint design}: choose the sampling (resolution, chroma, pre-filter) and the code together, for each image and bitrate \cite{kipnis2018}.
    \item \textbf{Aliasing inside learned codecs}: strided layers are sampling steps; alias-free design is known for generators \cite{karras2021}.
    \item \textbf{Continuous representations}: store a function and sample it at any resolution \cite{dupont2021}.
    \item \textbf{Perception}: where generated detail is acceptable, and how to signal it.
  \end{itemize}
\end{frame}

\section{Summary}

% 0:30
\begin{frame}{Takeaways}
  \begin{itemize}
    \item \textbf{Sampling}: with $f_s>2B$ the samples are enough and a sinc brings $x$ back. Below that: aliasing, for good.
    \item \textbf{Compression}: transform, quantize, entropy-code. DCT $\to$ wavelets $\to$ prediction $\to$ learned and generative codecs.
    \item \textbf{The link}: bits = samples $\times$ bits per sample, and the best split moves with the budget. The optimal lossy code drops the weak frequencies, so at low rates \alert{fewer samples are better}.
  \end{itemize}
\end{frame}
