\begin{frame}{能力与边界}{Agent 是加速器，而非领域专家}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{有参考即胜任}
        \begin{itemize}
          \item 阅读与修改代码、补充测试
          \item 生成样板与文档、格式化
          \item 运行命令、检索日志、总结与制表
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{缺少参考即不可靠}
        \begin{itemize}
          \item 编造参考文献
          \item 凭空实现领域规则
          \item 替代研究者作出主观判断
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \bigskip
  领域知识并不在其训练数据之中。使用 agent 的核心，是把领域知识转化为
  它能读的\structure{参考}，以及你能核验的\structure{输出}。

\section[外置规则]{外置领域规则}

% 0:50
\begin{frame}{通用实现与领域规则的冲突}
  \begin{block}{问题}
    Agent 的训练数据偏向教科书式的通用实现；领域规则通常是隐式且不成文的，
    不会出现在任何默认写法之中。
  \end{block}
  \begin{alertblock}{后果}
    缺乏参考时，模型自第一步（预处理）即偏离领域约定，其后的全部结果随之失效。
  \end{alertblock}
  \bigskip
  因此，第一步不是给它一个任务，而是给它一份\structure{可参考的领域约定}。
\end{frame}

% 0:50
\begin{frame}{维护一个领域 code base 仓库}
  \begin{block}{沉淀}
    将已验证的最小可运行代码、既往踩过的坑、约定的预处理流程，整理为一个仓库。
  \end{block}
  \begin{block}{用法}
    令 agent 以该仓库为参考修改代码，而非从零自由发挥。
  \end{block}
  \bigskip
  \begin{center}
  \textcolor{soft}{\small 有参考，agent 才有正确的模板可循；无参考，代码从第一步即错。}
  \end{center}
\end{frame}

% 1:00
\begin{frame}{案例：VAE 重建的三条规则}{AIGC 图像检测中的领域约定}
  \begin{center}
  \begin{tikzpicture}[>={Stealth[length=2mm]}, font=\footnotesize, node distance=4mm,
      box/.style={draw=#1, fill=#1!12, text=#1!75!black, rounded corners=2pt, minimum height=7mm, inner sep=4pt}]
    \node[box=soft] (img) {image};
    \node[box=accent, right=of img] (crop) {crop to 32x};
    \node[box=sig, right=of crop] (enc) {VAE encode};
    \node[box=sig, right=of enc] (dec) {VAE decode};
    \node[box=video, right=of dec] (png) {save PNG};
    \draw[thick, ->] (img) -- (crop);
    \draw[thick, ->] (crop) -- (enc);
    \draw[thick, ->] (enc) -- (dec);
    \draw[thick, ->] (dec) -- (png);
  \end{tikzpicture}
  \end{center}
  \begin{block}{三条规则}
    \begin{enumerate}
      \item 重建前将图像裁剪为长宽 32 的整数倍；否则长宽不整除时输出黑图或黑边。
      \item 禁止 \texttt{resize}：插值会改变伪影，而检测器对伪影高度敏感。
      \item 底层伪影以 PNG 保存：JPEG 的二次有损压缩会再次引入或抹除伪影。
    \end{enumerate}
  \end{block}
  \textcolor{soft}{\small 这些约定不在任何通用实现的默认中；无参考时，模型从预处理起即出错。}
\end{frame}

\section[落地材料]{落地原始材料}

% 0:50
\begin{frame}{调研中的参考文献幻觉}{编造，而非检索失败}
  \begin{block}{机制}
    Agent 无法下载或访问某篇文献时，通常不会报告失败，而是在「给出答案」
    这一后训练奖励的驱动下，生成一条难以察觉的伪造引用。
  \end{block}
  \begin{alertblock}{特征}
    伪造条目在作者、年份、venue 与编号上均合乎格式，不易识别。
  \end{alertblock}
  \bigskip
  对策见于下一页：自行下载，令其仅基于本地文件作答。
\end{frame}

% 0:50
\begin{frame}{先下载，再作答}
  \begin{block}{原则}
    自行将文献 PDF 下载至本地，令 agent 仅基于本地文件作答；
    无法访问时须如实说明，而非凭印象补全。
  \end{block}
  \begin{block}{工具}
    Zotero 统一管理论文（编号、标签、笔记），\texttt{llm-for-zotero}
    将本地库接入 agent。
  \end{block}
  \bigskip
  \begin{center}
  \textcolor{soft}{\small 落地原始材料，是切断编造入口最直接的手段。}
  \end{center}
\end{frame}

\section[显式验收]{显式验收}

% 0:50
\begin{frame}{Human in the loop}{每一步都需人工确认}
  \begin{center}
  \begin{tikzpicture}[>={Stealth[length=2mm]}, font=\footnotesize, node distance=4mm,
      box/.style={draw=#1, fill=#1!12, text=#1!75!black, rounded corners=2pt, minimum height=7mm, inner sep=4pt}]
    \node[box=sig] (you) {you set the rule};
    \node[box=accent, right=of you] (agent) {agent step};
    \node[box=sig, right=of agent] (verify) {you verify};
    \node[box=video, right=of verify] (next) {next step};
    \node[box=hot, below=8mm of agent] (fix) {fix and retry};
    \draw[thick, ->] (you) -- (agent);
    \draw[thick, ->] (agent) -- (verify);
    \draw[thick, ->] (verify) -- (next);
    \draw[thick, ->, hot] (verify.south) to[out=270, in=0] node[right, font=\scriptsize] {not OK} (fix.east);
    \draw[thick, ->, hot] (fix.west) to[out=180, in=270] (agent.south);
  \end{tikzpicture}
  \end{center}
  \begin{block}{要点}
    每一步须可验证——预处理是否正确、指标是否合理、基线是否复现。
    错误若发生在第一步，后续结果即全部失效。
  \end{block}
  \textcolor{soft}{\small 研究者才是判断「结果是否正确」的那个人。}
\end{frame}

% 0:50
\begin{frame}{定义完成，避免任务漂移}
  \begin{block}{拆解过程}
    将大目标拆为具有先后顺序的小步，每步产出一个可检查的产物。
  \end{block}
  \begin{block}{显式定义完成}
    明确完成后的形态、需运行的命令，以及不可改动的部分。
  \end{block}
  \begin{alertblock}{反例}
    未定义完成时，agent 会将「看起来完成」误判为完成，并持续偏离目标。
  \end{alertblock}
  \textcolor{soft}{\small 每完成一步，回到目标检查一次：是否仍在原路。}
\end{frame}

\section[低成本]{低成本运行}

% 0:50
\begin{frame}{低成本工作流}{opencode 与开源模型}
  \begin{block}{规划先行}
    opencode 的 \texttt{go plan} 先给出计划、经确认后执行，兼顾成本与方向。
  \end{block}
  \begin{block}{开源模型}
    性能与价格俱佳，足以覆盖约九成的工作（读改代码、生成样板、总结、查日志）；
    高成本模型仅用于算法设计与疑难缺陷。
  \end{block}
\end{frame}

% 0:40
\begin{frame}{subagent 与并行}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{并行}
        将无依赖的任务拆开并行执行，仅保留结论，不保留文件转储。
      \end{block}
      \begin{block}{判据}
        两任务是否存在先后依赖；无依赖即可并行。
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{center}
      \begin{tikzpicture}[font=\footnotesize, node distance=3mm,
          box/.style={draw=accent, fill=accent!10, text=accent!75!black, rounded corners=2pt, minimum height=7mm, minimum width=22mm, align=center, inner sep=3pt}]

      \end{center}

  \end{columns}
\end{frame}

\begin{frame}{自行适配所需工具}
  \begin{block}{做法}
    缺少适配时，\texttt{fork} 开源项目自行接入，而非等待上游支持。
  \end{block}
  \begin{block}{案例}
    \texttt{mejistus/llm-for-zotero}——为 \texttt{llm-for-zotero}
    增加 opencode 支持。
  \end{block}
  \bigskip
  \begin{center}
  \textcolor{soft}{\small 开源生态与 agent 编码的结合，可在较短时间内置所需工具。}
  \end{center}
\end{frame}
