\begin{frame}{主旨}
  \begin{center}
  {\Large Agent 能交付多少，取决于三件事是否就位：}\\[4pt]
  {\Large 领域规则是否\alert{外置}，}\\
  {\Large 原始材料是否\alert{落地}，}\\
  {\Large 每一步是否可\alert{验证}。}
  \end{center}
  \bigskip
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \textcolor{sig}{\textbf{外置规则}}\\[2pt]
      \small 让 agent 以领域约定为参考修改代码，而非自由发挥。
    \end{column}
    \begin{column}{0.32\textwidth}
      \textcolor{accent}{\textbf{落地材料}}\\[2pt]
      \small 让 agent 仅基于本地文献作答，而非凭记忆与检索。
    \end{column}
    \begin{column}{0.32\textwidth}
      \textcolor{video}{\textbf{显式验收}}\\[2pt]
      \small 让每一步都有可验证的完成标准，并由人逐一确认。
    \end{column}
  \end{columns}
  \bigskip
  \begin{center}
  \textbf{一句话：你能验证多少，Agent 就能做多少。}
  \end{center}
\end{frame}

% 0:40
\begin{frame}{能力与边界}{Agent 是加速器，而非领域专家}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{有参考即胜任}
        \begin{itemize}
          \item 阅读与修改代码、补充测试
          \item 生成样板与文档、格式化
          \item 运行命令、检索日志、总结与制表
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{缺少参考即不可靠}
        \begin{itemize}
          \item 编造参考文献
          \item 凭空实现领域规则
          \item 替代研究者作出主观判断
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \bigskip
  领域知识并不在其训练数据之中。使用 agent 的核心，是把领域知识转化为
  它能读的\structure{参考}，以及你能核验的\structure{输出}。
\end{frame}

\section[外置规则]{外置领域规则}

% 0:40
\begin{frame}{\textcolor{hot}{痛点}：代码从第一步就错}
  \begin{alertblock}{}
    让 agent 写一段 VAE 重建脚本，跑出来是黑图、黑边，或指标直接崩；
    改了三轮仍不对，因为它每次都从「教科书写法」重来——而教科书里没有你领域的规则。
  \end{alertblock}
  \bigskip
  你花在纠错上的时间，比它「省」下的还多；错在预处理这一步，后面的结果就全作废。
\end{frame}

% 0:50
\begin{frame}{通用实现与领域规则的冲突}
  \begin{block}{问题}
    Agent 的训练数据偏向教科书式的通用实现；领域规则通常是隐式且不成文的，
    不会出现在任何默认写法之中。
  \end{block}
  \begin{alertblock}{后果}
    缺乏参考时，模型自第一步（预处理）即偏离领域约定，其后的全部结果随之失效。
  \end{alertblock}
  \begin{block}{方案：维护一个领域 code base 仓库}
    将已验证的最小可运行代码、既往踩过的坑、约定的预处理流程整理为一个仓库，
    令 agent 以它为参考修改代码，而非从零自由发挥。
  \end{block}
\end{frame}

% 1:00
\begin{frame}{案例：VAE 重建的三条规则}{AIGC 图像检测中的领域约定}
  \begin{center}
  \begin{tikzpicture}[>={Stealth[length=2mm]}, font=\footnotesize, node distance=4mm,
      box/.style={draw=#1, fill=#1!12, text=#1!75!black, rounded corners=2pt, minimum height=7mm, inner sep=4pt}]
    \node[box=soft] (img) {image};
    \node[box=accent, right=of img] (crop) {crop to 32x};
    \node[box=sig, right=of crop] (enc) {VAE encode};
    \node[box=sig, right=of enc] (dec) {VAE decode};
    \node[box=video, right=of dec] (png) {save PNG};
    \draw[thick, ->] (img) -- (crop);
    \draw[thick, ->] (crop) -- (enc);
    \draw[thick, ->] (enc) -- (dec);
    \draw[thick, ->] (dec) -- (png);
  \end{tikzpicture}
  \end{center}
  \begin{block}{三条规则}
    \begin{enumerate}
      \item 重建前将图像裁剪为长宽 32 的整数倍；否则长宽不整除时输出黑图或黑边。
      \item 禁止 \texttt{resize}：插值会改变伪影，而检测器对伪影高度敏感。
      \item 底层伪影以 PNG 保存：JPEG 的二次有损压缩会再次引入或抹除伪影。
    \end{enumerate}
  \end{block}
\end{frame}

% 0:20
\begin{frame}{\textcolor{video}{最佳实践}：规则落仓库，agent 照着抄}
  \begin{exampleblock}{}
    把领域规则沉淀为 code base 仓库，令 agent 以之为参考——让它\structure{抄}，别让它\structure{想}；
    每一步由你验证。
  \end{exampleblock}
  \bigskip
  \begin{itemize}
    \item 维护 code base $\to$ 让它先读参考再改 $\to$ 每步验证（看张图 / 对指标 / 复现基线）。
  \end{itemize}
\end{frame}

\section[落地材料]{落地原始材料}

% 0:40
\begin{frame}{\textcolor{hot}{痛点}：综述里的引用是编的}
  \begin{alertblock}{}
    让 agent 写 Related Work，引用看似都对，点进去却发现某篇根本不存在，
    或作者、年份、venue 是拼的；关键数字也对不上。
  \end{alertblock}
  \bigskip
  最糟的是\alert{你没查的那几条}，很可能就是错的那几条。
\end{frame}

% 0:50
\begin{frame}{编造，而非检索失败}
  \begin{block}{机制}
    Agent 无法下载或访问某篇文献时，通常不会报告失败，而是在「给出答案」
    这一后训练奖励的驱动下，生成一条难以察觉的伪造引用。
  \end{block}
  \begin{alertblock}{特征}
    伪造条目在作者、年份、venue 与编号上均合乎格式，不易识别。
  \end{alertblock}
  \begin{block}{方案：先下载，再作答}
    自行将文献 PDF 下载至本地，令 agent 仅基于本地文件作答；Zotero 统一管理论文，
    \texttt{llm-for-zotero} 将本地库接入 agent。
  \end{block}
\end{frame}

% 0:20
\begin{frame}{\textcolor{video}{最佳实践}：先落 PDF，再让 agent 读}
  \begin{exampleblock}{}
    不给它真东西，就不让它开口引用；读不到就明说，绝不让它凭印象补全。
  \end{exampleblock}
  \bigskip
  \begin{itemize}
    \item 先下好 PDF $\to$ 喂本地路径 / 库 $\to$ 要求每条引用标注来源 $\to$ 抽查。
  \end{itemize}
\end{frame}

\section[显式验收]{显式验收}

% 0:40
\begin{frame}{\textcolor{hot}{痛点}：一次跑完，全错；goal 太大，跑偏}
  \begin{alertblock}{}
    让它一次跑完再一起看：错在第一步，后面全错；
    goal 给太大，做到一半偏了，还停不下来。
  \end{alertblock}
\end{frame}

% 0:50
\begin{frame}{Human in the loop}{每一步都需人工确认}
  \begin{center}
  \begin{tikzpicture}[>={Stealth[length=2mm]}, font=\footnotesize, node distance=4mm,
      box/.style={draw=#1, fill=#1!12, text=#1!75!black, rounded corners=2pt, minimum height=7mm, inner sep=4pt}]
    \node[box=sig] (you) {you set the rule};
    \node[box=accent, right=of you] (agent) {agent step};
    \node[box=sig, right=of agent] (verify) {you verify};
    \node[box=video, right=of verify] (next) {next step};
    \node[box=hot, below=8mm of agent] (fix) {fix and retry};
    \draw[thick, ->] (you) -- (agent);
    \draw[thick, ->] (agent) -- (verify);
    \draw[thick, ->] (verify) -- (next);
    \draw[thick, ->, hot] (verify.south) to[out=270, in=0] node[right, font=\scriptsize] {not OK} (fix.east);
    \draw[thick, ->, hot] (fix.west) to[out=180, in=270] (agent.south);
  \end{tikzpicture}
  \end{center}
  \begin{block}{要点}
    每一步须可验证——预处理是否正确、指标是否合理、基线是否复现；
    研究者才是判断「结果是否正确」的那个人。
  \end{block}
\end{frame}

% 0:40
\begin{frame}{定义完成，避免任务漂移}
  \begin{block}{拆解过程}
    将大目标拆为具有先后顺序的小步，每步产出一个可检查的产物。
  \end{block}
  \begin{block}{显式定义完成}
    明确完成后的形态、需运行的命令，以及不可改动的部分。
  \end{block}
  \begin{alertblock}{反例}
    未定义完成时，agent 会将「看起来完成」误判为完成，并持续偏离目标。
  \end{alertblock}
  \textcolor{soft}{\small 每完成一步，回到目标检查一次：是否仍在原路。}
\end{frame}

% 0:20
\begin{frame}{\textcolor{video}{最佳实践}：每步可验证，完成有定义}
  \begin{exampleblock}{}
    每一步都须可验证并由你确认；同时显式定义「完成」——做完什么样、跑什么、别动什么。
  \end{exampleblock}
  \bigskip
  \begin{itemize}
    \item 拆解 $\to$ 定义完成 $\to$ 每步回到目标检查一次。
  \end{itemize}
\end{frame}

\section[低成本]{低成本运行}

% 0:30
\begin{frame}{\textcolor{hot}{痛点}：太贵、太慢、还跑偏}
  \begin{alertblock}{}
    全用最贵的模型，账单吓人，却大多用不上；活是串行做的，慢；goal 太大，跑偏。
  \end{alertblock}
\end{frame}

% 0:50
\begin{frame}{低成本工作流}{opencode、开源模型与并行}
  \begin{block}{规划先行 + 开源模型}
    opencode 的 \texttt{go plan} 先给计划、经确认后执行，兼顾成本与方向；
    开源模型性能与价格俱佳，足以覆盖约九成工作，高成本模型仅用于算法设计与疑难缺陷。
  \end{block}
  \begin{block}{subagent 并行}
    将无依赖的任务拆开并行执行，仅保留结论，不保留文件转储；无依赖即可并行。
  \end{block}
  \begin{block}{自行适配工具}
    缺适配时 \texttt{fork} 开源项目自行接入——例 \texttt{mejistus/llm-for-zotero}
    为 \texttt{llm-for-zotero} 增加 opencode 支持。
  \end{block}
\end{frame}

% 0:20
\begin{frame}{\textcolor{video}{最佳实践}：go plan + 并行 + 拆解}
  \begin{exampleblock}{}
    go plan 先定方向、subagent 并行、goal 拆小并定义完成；贵模型只留给最难的一成。
  \end{exampleblock}
  \bigskip
  \begin{itemize}
    \item go plan 定方向 $\to$ 能并行就并行 $\to$ 每步回到 goal 检查。
  \end{itemize}
\end{frame}
